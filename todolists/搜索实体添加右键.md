- 右键实体弹出紧凑菜单，菜单包括：时间、名称（可编辑）、经纬度、航向、航速、图标（可更换）、删除、保存天数（可修改，默认2天）

- 支持保存实体到 localStorage，保存天数可自定义

<script setup lang="ts">
import { ref, reactive, watch, onMounted, onUnmounted, nextTick } from 'vue';
import * as Cesium from 'cesium';

const props = defineProps<{ viewer: Cesium.Viewer }>();

// 图标列表（可自定义）
const iconList = [
  { label: '默认', value: '' },
  { label: '飞机', value: '/icons/plane.png' },
  { label: '船', value: '/icons/ship.png' },
  { label: '车', value: '/icons/car.png' },
];

// 右键菜单相关
const menuVisible = ref(false);
const menuPosition = reactive({ x: 0, y: 0 });
const selectedEntity = ref<Cesium.Entity | null>(null);
const selectedEntityData = reactive({
  id: '',
  time: '',
  name: '',
  lon: 0,
  lat: 0,
  heading: 0,
  speed: 0,
  icon: '',
  keepDays: 2,
});

// 实体管理
const entityPrefix = 'flash-point-';
const storageKey = 'cesium-entities';

function getNowTimeStr() {
  const d = new Date();
  return `${d.getFullYear()}-${d.getMonth() + 1}-${d.getDate()} ${d.getHours()}:${d.getMinutes()}:${d.getSeconds()}`;
}

// 创建实体（外部可调用）
function createEntity({ lon, lat, heading = 0, speed = 0, name = '', icon = '', keepDays = 2 }) {
  const id = entityPrefix + Date.now() + Math.floor(Math.random() * 10000);
  const time = getNowTimeStr();
  const entity = props.viewer.entities.add({
    id,
    position: Cesium.Cartesian3.fromDegrees(lon, lat, 0),
    billboard: icon
      ? {
          image: icon,
          width: 32,
          height: 32,
          heightReference: Cesium.HeightReference.CLAMP_TO_GROUND,
        }
      : undefined,
    point: !icon
      ? {
          pixelSize: 16,
          color: Cesium.Color.YELLOW.withAlpha(0.9),
          outlineColor: Cesium.Color.RED,
          outlineWidth: 4,
          heightReference: Cesium.HeightReference.CLAMP_TO_GROUND,
        }
      : undefined,
    polyline: {
      positions: [
        Cesium.Cartesian3.fromDegrees(lon, lat, 0),
        Cesium.Cartesian3.fromDegrees(lon, lat, 30),
      ],
      width: 2,
      material: Cesium.Color.ORANGE,
      clampToGround: true,
    },
    label: {
      show: true,
      text: buildLabelText({ time, name, lon, lat, heading, speed }),
      font: '14px sans-serif',
      fillColor: Cesium.Color.BLACK,
      style: Cesium.LabelStyle.FILL,
      verticalOrigin: Cesium.VerticalOrigin.BOTTOM,
      pixelOffset: new Cesium.Cartesian2(0, -40),
      heightReference: Cesium.HeightReference.CLAMP_TO_GROUND,
      backgroundColor: Cesium.Color.WHITE.withAlpha(0.7),
      showBackground: true,
    },
    properties: {
      time,
      name,
      lon,
      lat,
      heading,
      speed,
      icon,
      keepDays,
    },
  });
  saveEntityToStorage(entity);
  return entity;
}

// 构建标牌内容
function buildLabelText({ time, name, lon, lat, heading, speed }: any) {
  return `时间:${time}\n名称:${name}\n经度:${lon.toFixed(6)}\n纬度:${lat.toFixed(6)}\n航向:${heading}\n航速:${speed}`;
}

// 右键菜单事件
function onRightClick(movement: any) {
  const picked = props.viewer.scene.pick(movement.position);
  if (Cesium.defined(picked) && picked.id && picked.id.id?.startsWith(entityPrefix)) {
    selectedEntity.value = picked.id;
    const p = picked.id;
    // 读取属性
    selectedEntityData.id = p.id;
    selectedEntityData.time = p.properties.time.getValue();
    selectedEntityData.name = p.properties.name.getValue();
    selectedEntityData.lon = p.properties.lon.getValue();
    selectedEntityData.lat = p.properties.lat.getValue();
    selectedEntityData.heading = p.properties.heading.getValue();
    selectedEntityData.speed = p.properties.speed.getValue();
    selectedEntityData.icon = p.properties.icon.getValue();
    selectedEntityData.keepDays = p.properties.keepDays.getValue();
    menuPosition.x = movement.position.x;
    menuPosition.y = movement.position.y;
    menuVisible.value = true;
  } else {
    menuVisible.value = false;
    selectedEntity.value = null;
  }
}

// 保存实体到本地
function saveEntityToStorage(entity: Cesium.Entity) {
  const arr = loadEntitiesFromStorage();
  // 先移除同id
  const idx = arr.findIndex((e: any) => e.id === entity.id);
  if (idx !== -1) arr.splice(idx, 1);
  arr.push({
    id: entity.id,
    time: entity.properties.time.getValue(),
    name: entity.properties.name.getValue(),
    lon: entity.properties.lon.getValue(),
    lat: entity.properties.lat.getValue(),
    heading: entity.properties.heading.getValue(),
    speed: entity.properties.speed.getValue(),
    icon: entity.properties.icon.getValue(),
    keepDays: entity.properties.keepDays.getValue(),
    created: Date.now(),
  });
  localStorage.setItem(storageKey, JSON.stringify(arr));
}

// 加载所有未过期实体
function loadEntitiesFromStorage() {
  const arr = JSON.parse(localStorage.getItem(storageKey) || '[]');
  const now = Date.now();
  return arr.filter((e: any) => now - e.created < (e.keepDays || 2) * 24 * 3600 * 1000);
}

// 恢复实体
function restoreEntities() {
  const arr = loadEntitiesFromStorage();
  arr.forEach((e: any) => {
    createEntity(e);
  });
}

// 保存按钮
function saveEntity() {
  if (!selectedEntity.value) return;
  // 更新实体属性
  selectedEntity.value.properties.name = selectedEntityData.name;
  selectedEntity.value.properties.lon = selectedEntityData.lon;
  selectedEntity.value.properties.lat = selectedEntityData.lat;
  selectedEntity.value.properties.heading = selectedEntityData.heading;
  selectedEntity.value.properties.speed = selectedEntityData.speed;
  selectedEntity.value.properties.icon = selectedEntityData.icon;
  selectedEntity.value.properties.keepDays = selectedEntityData.keepDays;
  // 更新位置
  selectedEntity.value.position = Cesium.Cartesian3.fromDegrees(selectedEntityData.lon, selectedEntityData.lat, 0);
  // 更新图标
  if (selectedEntityData.icon) {
    selectedEntity.value.billboard = {
      image: selectedEntityData.icon,
      width: 32,
      height: 32,
      heightReference: Cesium.HeightReference.CLAMP_TO_GROUND,
    };
    selectedEntity.value.point = undefined;
  } else {
    selectedEntity.value.billboard = undefined;
    selectedEntity.value.point = {
      pixelSize: 16,
      color: Cesium.Color.YELLOW.withAlpha(0.9),
      outlineColor: Cesium.Color.RED,
      outlineWidth: 4,
      heightReference: Cesium.HeightReference.CLAMP_TO_GROUND,
    };
  }
  // 更新引线
  selectedEntity.value.polyline = {
    positions: [
      Cesium.Cartesian3.fromDegrees(selectedEntityData.lon, selectedEntityData.lat, 0),
      Cesium.Cartesian3.fromDegrees(selectedEntityData.lon, selectedEntityData.lat, 30),
    ],
    width: 2,
    material: Cesium.Color.ORANGE,
    clampToGround: true,
  };
  // 更新标牌
  selectedEntity.value.label = {
    show: true,
    text: buildLabelText(selectedEntityData),
    font: '14px sans-serif',
    fillColor: Cesium.Color.BLACK,
    style: Cesium.LabelStyle.FILL,
    verticalOrigin: Cesium.VerticalOrigin.BOTTOM,
    pixelOffset: new Cesium.Cartesian2(0, -40),
    heightReference: Cesium.HeightReference.CLAMP_TO_GROUND,
    backgroundColor: Cesium.Color.WHITE.withAlpha(0.7),
    showBackground: true,
  };
  saveEntityToStorage(selectedEntity.value);
  menuVisible.value = false;
}

// 删除按钮
function deleteEntity() {
  if (!selectedEntity.value) return;
  props.viewer.entities.remove(selectedEntity.value);
  // 从本地删除
  const arr = loadEntitiesFromStorage();
  const idx = arr.findIndex((e: any) => e.id === selectedEntity.value.id);
  if (idx !== -1) arr.splice(idx, 1);
  localStorage.setItem(storageKey, JSON.stringify(arr));
  menuVisible.value = false;
  selectedEntity.value = null;
}

function closeMenu() {
  menuVisible.value = false;
}

// 监听右键
let handler: Cesium.ScreenSpaceEventHandler;
onMounted(() => {
  restoreEntities();
  handler = new Cesium.ScreenSpaceEventHandler(props.viewer.scene.canvas);
  handler.setInputAction(onRightClick, Cesium.ScreenSpaceEventType.RIGHT_CLICK);
});
onUnmounted(() => {
  handler?.destroy();
});
</script>

## 用法说明

1. 在 `CesiumMap.vue` 中引入并传入 `viewer` 实例：

```vue
<EntityManager :viewer="viewer" />
```

2. 通过调用 `createEntity({ lon, lat, heading, speed, name, icon, keepDays })` 方法创建实体（可在 geocoder 定位后调用）。
3. 右键实体弹出菜单，可编辑所有信息、选择图标、设置保存天数、删除、保存。
4. 实体有引线和标牌，标牌显示所有关键信息。
5. 实体信息自动保存到 localStorage，默认保存2天，可自定义。
